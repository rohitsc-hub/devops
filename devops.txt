Script code:
pipeline {
    agent any


    tools {
        maven 'Maven'
    }


    stages {
        stage('Create Maven Project') {
            steps {
                echo 'Creating Java Maven project...'


                bat 'if not exist src\\main\\java mkdir src\\main\\java'


                writeFile file: 'pom.xml', text: '''<project xmlns="http://maven.apache.org/POM/4.0.0"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xsi:schemaLocation="http://maven.apache.org/POM/4.0.0 https://maven.apache.org/xsd/maven-4.0.0.xsd">
    <modelVersion>4.0.0</modelVersion>
    <groupId>com.example</groupId>
    <artifactId>myapp</artifactId>
    <version>1.0</version>
    <properties>
        <maven.compiler.source>8</maven.compiler.source>
        <maven.compiler.target>8</maven.compiler.target>
        <project.build.sourceEncoding>UTF-8</project.build.sourceEncoding>
    </properties>
</project>
'''


                writeFile file: 'src/main/java/App.java', text: '''public class App {
    public static void main(String[] args) {
        System.out.println("Hello from Maven and Jenkins!");
    }
}
'''
            }
        }


        stage('Build') {
            steps {
                echo 'Building Java application using Maven...'
                bat 'mvn -version'
                bat 'mvn clean package'
            }
        }


        stage('Test') {
            steps {
                echo 'Checking generated JAR file...'
                bat 'dir target'
            }
        }


        stage('Deploy') {
            steps {
                archiveArtifacts artifacts: 'target/*.jar',
                                 allowEmptyArchive: false


                echo 'Java application built and archived successfully!'
            }
        }
    }
}




Build now> builds>console output

            steps {

app.py:
import os
import time
import pymysql
from flask import Flask, jsonify


app = Flask(__name__)


def get_db_connection():
    return pymysql.connect(
        host=os.getenv("DB_HOST", "db"),
        user=os.getenv("DB_USER", "appuser"),
        password=os.getenv("DB_PASSWORD", "apppassword"),
        database=os.getenv("DB_NAME", "appdb"),
        connect_timeout=5
    )


@app.route("/")
def home():
    return """
    <h1>Docker Compose Practical</h1>
    <p>Webserver container is running!</p>
    <p><a href="/health">Check database connection</a></p>
    """


@app.route("/health")
def health():
    try:
        connection = get_db_connection()
        with connection.cursor() as cursor:
            cursor.execute("SELECT 1")
            result = cursor.fetchone()
        connection.close()
        return jsonify(status="healthy", database="connected", result=result[0])
    except Exception as error:
        return jsonify(status="unhealthy", error=str(error)), 503


if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)

Compose.yaml:


services:
  web:
    build: .
    container_name: practical-web
    ports:
      - "5000:5000"
    environment:
      DB_HOST: db
      DB_USER: appuser
      DB_PASSWORD: ${DB_PASSWORD:-apppassword}
      DB_NAME: appdb
    depends_on:
      db:
        condition: service_healthy
    restart: unless-stopped


  db:
    image: mysql:8.0
    container_name: practical-db
    environment:
      MYSQL_ROOT_PASSWORD: ${MYSQL_ROOT_PASSWORD:-rootpassword}
      MYSQL_DATABASE: appdb
      MYSQL_USER: appuser
      MYSQL_PASSWORD: ${DB_PASSWORD:-apppassword}
    volumes:
      - mysql_data:/var/lib/mysql
    healthcheck:
      test: ["CMD", "mysqladmin", "ping", "-h", "localhost", "-uroot", "-p$${MYSQL_ROOT_PASSWORD}"]
      interval: 10s
      timeout: 5s
      retries: 10
      start_period: 30s
